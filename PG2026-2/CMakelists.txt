cmake_minimum_required(VERSION 3.11)
project(PG2026-2)

# Define o padrão do C++
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED True)

# Ativa o FetchContent
include(FetchContent)

# Configura o FetchContent para baixar e compilar GLFW automaticamente
FetchContent_Declare(
  glfw
  GIT_REPOSITORY https://github.com/glfw/glfw.git
  GIT_TAG master  # Define a versão desejada da GLFW
  GIT_SHALLOW TRUE
)
FetchContent_MakeAvailable(glfw)

FetchContent_Declare(
  glm
  GIT_REPOSITORY https://github.com/g-truc/glm.git
  GIT_TAG master
  GIT_SHALLOW TRUE
)
FetchContent_MakeAvailable(glm)

# Configura o FetchContent para baixar a stb_image automaticamente
FetchContent_Declare(
  stb_image
  GIT_REPOSITORY https://github.com/nothings/stb.git
  GIT_TAG master
  GIT_SHALLOW TRUE
)
FetchContent_MakeAvailable(stb_image)

# Adiciona as pastas de cabeçalhos
include_directories(${CMAKE_SOURCE_DIR}/include)
include_directories(${CMAKE_SOURCE_DIR}/common)
include_directories(${CMAKE_SOURCE_DIR}/include/glad)
include_directories(${glm_SOURCE_DIR})
include_directories(${stb_image_SOURCE_DIR})

# Lista de pastas de exercícios/projetos
# OBS.: Não pode ter acentos ou caracteres especiais
set(EXERCISES
    Exemplos/HelloTriangle
    Lista1
    Lista2
)

add_compile_options(-Wno-pragmas)

# Define as bibliotecas para cada sistema operacional
if(WIN32)
    set(OPENGL_LIBS opengl32)
elseif(APPLE)
    find_library(OpenGL_LIBRARY OpenGL)
    set(OPENGL_LIBS ${OpenGL_LIBRARY})
else()
    find_package(OpenGL REQUIRED)
    set(OPENGL_LIBS ${OPENGL_gl_LIBRARY})
endif()

# Caminho esperado para a GLAD
set(GLAD_C_FILE "${CMAKE_SOURCE_DIR}/common/glad.c")

# Verifica se os arquivos da GLAD estão no lugar
if (NOT EXISTS ${GLAD_C_FILE})
    message(FATAL_ERROR "Arquivo glad.c não encontrado! 
    Baixe a GLAD manualmente em https://glad.dav1d.de/ 
    e coloque glad.h em include/glad/ e glad.c em common/")
endif()

foreach(EXERCISE ${EXERCISES})
    # Coleta todos os .cpp dentro do diretório do exercício
    # CONFIGURE_DEPENDS faz o CMake reexecutar a busca quando um .cpp é criado/removido
    file(GLOB EXE_SOURCES CONFIGURE_DEPENDS src/${EXERCISE}/*.cpp)

    # Separa os arquivos que têm main() dos arquivos de apoio
    set(MAIN_SOURCES "")
    set(SUPPORT_SOURCES "")
    foreach(SOURCE ${EXE_SOURCES})
        file(READ ${SOURCE} SOURCE_CONTENT)
        if(SOURCE_CONTENT MATCHES "int[ \t\r\n]+main[ \t\r\n]*\\(")
            list(APPEND MAIN_SOURCES ${SOURCE})
        else()
            list(APPEND SUPPORT_SOURCES ${SOURCE})
        endif()
    endforeach()

    if(NOT MAIN_SOURCES)
        message(WARNING "Nenhum arquivo com main() encontrado em src/${EXERCISE}")
    endif()

    # Nome da pasta do exercício (ex.: Lista1, Lista2, HelloTriangle)
    get_filename_component(EXERCISE_DIR ${EXERCISE} NAME)

    foreach(MAIN_SOURCE ${MAIN_SOURCES})
        # Extrai o nome do arquivo (sem extensão) como nome do executável
        get_filename_component(EXE_NAME ${MAIN_SOURCE} NAME_WE)

        # O alvo CMake recebe a pasta como prefixo (ex.: Lista2_exercicio1) para que
        # arquivos de mesmo nome em listas diferentes não entrem em conflito
        set(TARGET_NAME ${EXERCISE_DIR}_${EXE_NAME})

        # Cria o executável com o .cpp do main + arquivos de apoio + glad.c
        add_executable(${TARGET_NAME} ${MAIN_SOURCE} ${SUPPORT_SOURCES} ${GLAD_C_FILE})

        # O binário mantém o nome do .cpp e é gerado em build/<pasta>/ (ex.: build/Lista2/exercicio1)
        set_target_properties(${TARGET_NAME} PROPERTIES
            OUTPUT_NAME ${EXE_NAME}
            RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/${EXERCISE_DIR})

        # Configura includes e bibliotecas
        target_include_directories(${TARGET_NAME} PRIVATE ${CMAKE_SOURCE_DIR}/include/glad ${glm_SOURCE_DIR} ${stb_image_SOURCE_DIR})

        target_link_libraries(${TARGET_NAME} glfw ${OPENGL_LIBS} glm::glm )

        message(STATUS "Executável: ${EXERCISE_DIR}/${EXE_NAME} (alvo ${TARGET_NAME})")
    endforeach()
endforeach()
